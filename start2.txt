unzip ig-comment-bot.zip
cd ig-comment-bot
npm install
cp .env.example .env    # заполните META_APP_ID/SECRET, OAUTH_REDIRECT_URI и т.д.

npm run dev       # терминал 1
npm run worker    # терминал 2

cd frontend
npm install
cp .env.example .env
npm run dev        # терминал 3 → http://localhost:5173
